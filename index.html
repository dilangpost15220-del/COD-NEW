① เชื่อมต่อ Google Apps Script (Google Sheet)
ตั้งค่า Web App URL เพื่อ sync ข้อมูลกับ Google Sheet — ต้อง Deploy เป็น Execute as: Me, Who has access: Anyone
ต้องขึ้นต้นด้วย https://script.google.com/macros/s/ และลงท้ายด้วย /exec
📋 โค้ด Google Apps Script ที่จำเป็น: 
// ติดตั้งใน Google Apps Script แล้ว Deploy เป็น Web App
function doGet(e) {
  const action = e.parameter.action;
  if (action === 'test') return ContentService.createTextOutput(JSON.stringify({status:'success',message:'Connected!'})).setMimeType(ContentService.MimeType.JSON);
  if (action === 'getAll') return getAll();
  return ContentService.createTextOutput(JSON.stringify({status:'error',message:'Unknown action'})).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  if (body.action === 'saveBatch') return saveBatch(body.data);
  if (body.action === 'uploadPhoto') return uploadPhoto(body);
  return ContentService.createTextOutput(JSON.stringify({status:'error'})).setMimeType(ContentService.MimeType.JSON);
}
② อัปโหลดรูปภาพผ่าน Google Drive (Apps Script)
รูปภาพจะถูกอัปโหลดผ่าน Apps Script เดียวกัน ไม่ต้องใช้ OAuth แยก
รูปภาพจะถูกส่งผ่าน Apps Script URL เดียวกับ Google Sheet — ตรวจสอบว่าสคริปต์มีสิทธิ์ Drive
เปิด Google Drive Folder → URL จะมี /folders/FOLDER_ID — คัดลอกมาใส่ที่นี่
③ อัปโหลดรูปภาพผ่าน GitHub (ทางเลือก)
ฟรี — ไม่ต้องใช้ Google Drive — รูปเก็บใน Public Repository
ต้องมีสิทธิ์ repo — สร้างได้ที่ GitHub → Settings → Developer Settings → Personal Access Tokens
④ จัดการข้อมูลในเครื่อง (LocalStorage)
สำรอง / กู้คืน / ล้างข้อมูลทั้งหมด
